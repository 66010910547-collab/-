    <!-- Logic Script -->
    <script>
        const apiKey = ""; // Canvas runtime provides API Key automatically

        // Element References
        const cameraInput = document.getElementById('cameraInput');
        const galleryInput = document.getElementById('galleryInput');
        const emptyPlaceholder = document.getElementById('emptyPlaceholder');
        const imageGrid = document.getElementById('imageGrid');
        const analyzeBtn = document.getElementById('analyzeBtn');

        const loadingSection = document.getElementById('loadingSection');
        const resultSection = document.getElementById('resultSection');
        const ingredientsList = document.getElementById('ingredientsList');
        const recipesGrid = document.getElementById('recipesGrid');

        const guideBtn = document.getElementById('guideBtn');
        const guideSection = document.getElementById('guideSection');
        const closeGuideBtn = document.getElementById('closeGuideBtn');

        const recipeModal = document.getElementById('recipeModal');
        const closeModalBtn = document.getElementById('closeModalBtn');
        const closeModalBottomBtn = document.getElementById('closeModalBottomBtn');

        // Store selected images base64 data
        let imageList = []; // Array of objects: { id, base64, dataUrl }
        let currentRecipes = [];

        // Guide Modal Toggle
        guideBtn.addEventListener('click', () => guideSection.classList.toggle('hidden'));
        closeGuideBtn.addEventListener('click', () => guideSection.classList.add('hidden'));

        // Handle Camera Capture Input
        cameraInput.addEventListener('change', (e) => {
            if (e.target.files && e.target.files[0]) {
                processFiles([e.target.files[0]]);
                e.target.value = ''; // reset input
            }
        });

        // Handle Gallery Selection Input
        galleryInput.addEventListener('change', (e) => {
            if (e.target.files && e.target.files.length > 0) {
                processFiles(Array.from(e.target.files));
                e.target.value = ''; // reset input
            }
        });

        // Process File List to Base64
        function processFiles(files) {
            files.forEach(file => {
                const reader = new FileReader();
                reader.onload = (event) => {
                    const dataUrl = event.target.result;
                    const base64 = dataUrl.split(',')[1];
                    const id = Date.now() + Math.random().toString(36).substring(2, 9);
                    
                    imageList.push({ id, base64, dataUrl });
                    renderImageGrid();
                };
                reader.readAsDataURL(file);
            });
        }

        // Render Image Grid Thumbnails
        function renderImageGrid() {
            if (imageList.length === 0) {
                emptyPlaceholder.classList.remove('hidden');
                imageGrid.classList.add('hidden');
                analyzeBtn.disabled = true;
                return;
            }

            emptyPlaceholder.classList.add('hidden');
            imageGrid.classList.remove('hidden');
            analyzeBtn.disabled = false;

            imageGrid.innerHTML = '';
            imageList.forEach((imgObj) => {
                const card = document.createElement('div');
                card.className = "relative group aspect-square rounded-lg overflow-hidden bg-slate-200 border border-slate-300 shadow-sm";
                card.innerHTML = `
                    <img src="${imgObj.dataUrl}" class="w-full h-full object-cover" alt="วัตถุดิบ" />
                    <button onclick="removeImage('${imgObj.id}')" class="absolute top-1 right-1 bg-red-500 hover:bg-red-600 text-white rounded-full w-6 h-6 flex items-center justify-center text-xs shadow-md transition">
                        <i class="fa-solid fa-xmark"></i>
                    </button>
                `;
                imageGrid.appendChild(card);
            });
        }

        // Remove Single Image
        window.removeImage = function(id) {
            imageList = imageList.filter(img => img.id !== id);
            renderImageGrid();
        };

        // Analyze Images via Gemini API
        analyzeBtn.addEventListener('click', async () => {
            if (imageList.length === 0) return;

            loadingSection.classList.remove('hidden');
            resultSection.classList.add('hidden');
            analyzeBtn.disabled = true;

            try {
                const responseData = await fetchRecipeFromGemini(imageList);
                displayResults(responseData);
            } catch (error) {
                alert("เกิดข้อผิดพลาดในการวิเคราะห์รูปภาพ: " + error.message);
            } finally {
                loadingSection.classList.add('hidden');
                analyzeBtn.disabled = false;
            }
        });

        // Gemini API Interaction Function
        async function fetchRecipeFromGemini(images) {
            const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

            const promptText = `
            ช่วยวิเคราะห์ภาพวัตถุดิบอาหารทั้งหมดที่แนบมานี้ จากนั้นระบุรายการวัตถุดิบที่ตรวจพบ และเสนอเมนูอาหารไทยหรืออาหารยอดนิยมที่ทำได้จากวัตถุดิบเหล่านี้มา 3-4 เมนู
            ตอบกลับเป็นรูปแบบ JSON ตามโครงสร้างนี้เท่านั้น:
            {
                "detectedIngredients": ["วัตถุดิบ 1", "วัตถุดิบ 2"],
                "recipes": [
                    {
                        "id": "1",
                        "title": "ชื่อเมนูอาหาร",
                        "category": "ประเภทอาหาร (เช่น ผัด, ต้ม, ทอด)",
                        "prepTime": "ระยะเวลาทำ (เช่น 15 นาที)",
                        "difficulty": "ระดับความยาก (ง่าย/ปานกลาง/ยาก)",
                        "description": "คำอธิบายชวนรับประทานเกี่ยวกับเมนูนี้",
                        "ingredients": ["ส่วนผสมทั้งหมดรวมถึงเครื่องปรุงที่ต้องใช้เพิ่มเติม"],
                        "steps": ["ขั้นตอนที่ 1...", "ขั้นตอนที่ 2..."]
                    }
                ]
            }
            `;

            // Construct payload parts array with prompt and images
            const parts = [{ text: promptText }];
            images.forEach(img => {
                parts.push({
                    inlineData: {
                        mimeType: "image/jpeg",
                        data: img.base64
                    }
                });
            });

            const payload = {
                contents: [{ role: "user", parts: parts }],
                generationConfig: { responseMimeType: "application/json" }
            };

            const delays = [1000, 2000, 4000, 8000, 16000];
            for (let i = 0; i <= delays.length; i++) {
                try {
                    const res = await fetch(endpoint, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!res.ok) throw new Error(`HTTP Error Status: ${res.status}`);

                    const data = await res.json();
                    const textResult = data.candidates?.[0]?.content?.parts?.[0]?.text;
                    if (textResult) {
                        return JSON.parse(textResult);
                    } else {
                        throw new Error("ไม่สามารถอ่านผลลัพธ์จาก AI ได้");
                    }
                } catch (err) {
                    if (i === delays.length) throw err;
                    await new Promise(resolve => setTimeout(resolve, delays[i]));
                }
            }
        }

        // Render Results
        function displayResults(data) {
            ingredientsList.innerHTML = '';
            recipesGrid.innerHTML = '';

            // Render Ingredients
            if (data.detectedIngredients && data.detectedIngredients.length > 0) {
                data.detectedIngredients.forEach(item => {
                    const badge = document.createElement('span');
                    badge.className = "bg-orange-50 text-orange-600 border border-orange-200 text-xs px-3 py-1 rounded-full font-medium flex items-center gap-1";
                    badge.innerHTML = `<i class="fa-solid fa-check text-xs"></i> ${item}`;
                    ingredientsList.appendChild(badge);
                });
            } else {
                ingredientsList.innerHTML = `<span class="text-xs text-slate-400">ไม่พบวัตถุดิบที่ระบุได้</span>`;
            }

            // Render Recipe Cards
            currentRecipes = data.recipes || [];
            if (currentRecipes.length > 0) {
                currentRecipes.forEach((recipe, index) => {
                    const card = document.createElement('div');
                    card.className = "bg-white border border-slate-200 rounded-xl p-4 hover:shadow-md transition duration-200 flex flex-col justify-between";
                    card.innerHTML = `
                        <div>
                            <div class="flex justify-between items-start mb-2">
                                <span class="text-xs font-semibold text-amber-600 bg-amber-50 px-2 py-0.5 rounded">${recipe.category || 'เมนูทั่วไป'}</span>
                                <span class="text-xs text-slate-400"><i class="fa-regular fa-clock"></i> ${recipe.prepTime || 'N/A'}</span>
                            </div>
                            <h4 class="font-bold text-slate-800 text-base mb-1">${recipe.title}</h4>
                            <p class="text-slate-500 text-xs line-clamp-2 mb-4">${recipe.description}</p>
                        </div>
                        <button onclick="openRecipeModal(${index})" class="w-full bg-amber-50 hover:bg-amber-100 text-amber-700 font-medium text-xs py-2 rounded-lg transition duration-150 flex items-center justify-center gap-1">
                            <i class="fa-solid fa-utensils text-xs"></i> ดูสูตรและวิธีทำ
                        </button>
                    `;
                    recipesGrid.appendChild(card);
                });
            } else {
                recipesGrid.innerHTML = `<p class="text-sm text-slate-500 col-span-2 text-center py-4">ไม่พบเมนูแนะนำจากวัตถุดิบที่ระบุ</p>`;
            }

            resultSection.classList.remove('hidden');
            resultSection.scrollIntoView({ behavior: 'smooth' });
        }

        // Open Detail Modal
        window.openRecipeModal = function(index) {
            const recipe = currentRecipes[index];
            if (!recipe) return;

            document.getElementById('modalTitle').textContent = recipe.title;
            document.getElementById('modalCategory').textContent = recipe.category || 'อาหารทั่วไป';
            document.getElementById('modalPrepTime').textContent = recipe.prepTime || 'ไม่ระบุ';
            document.getElementById('modalDifficulty').textContent = recipe.difficulty || 'ปานกลาง';
            document.getElementById('modalDescription').textContent = recipe.description || '';

            const modalIngredients = document.getElementById('modalIngredients');
            modalIngredients.innerHTML = '';
            (recipe.ingredients || []).forEach(ing => {
                const li = document.createElement('li');
                li.textContent = ing;
                modalIngredients.appendChild(li);
            });

            const modalSteps = document.getElementById('modalSteps');
            modalSteps.innerHTML = '';
            (recipe.steps || []).forEach((step, idx) => {
                const li = document.createElement('li');
                li.className = "flex gap-3 items-start";
                li.innerHTML = `
                    <span class="bg-amber-500 text-white font-bold rounded-full w-5 h-5 flex items-center justify-center text-xs shrink-0 mt-0.5">${idx + 1}</span>
                    <span>${step}</span>
                `;
                modalSteps.appendChild(li);
            });

            recipeModal.classList.remove('hidden');
        };

        // Modal Close Listeners
        closeModalBtn.addEventListener('click', () => recipeModal.classList.add('hidden'));
        closeModalBottomBtn.addEventListener('click', () => recipeModal.classList.add('hidden'));
        recipeModal.addEventListener('click', (e) => {
            if (e.target === recipeModal) recipeModal.classList.add('hidden');
        });

    </script>
